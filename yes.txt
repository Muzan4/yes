using System;

class Program

{

    static void Main()

    {

        Console.Write("Enter Number 1: ");

        int a = int.Parse(Console.ReadLine());

Practical 8: Session & TempData State Management
Steps to Do It:
Create an ASP.NET Core MVC project.
Step 1 (Startup.cs): Enable Session in ConfigureServices and Configure.
Step 2 (HomeController.cs): Add methods to Set, Get, and Clear Session and TempData.
Step 3 (Views/Home/): Create the corresponding .cshtml views (SetSessionData, GetSessionData, SetTempData, GetTempData).
Code:
1. Startup.cs
public void ConfigureServices(IServiceCollection services)

{

    services.AddSession(); // <-- ADD THIS

    services.AddControllersWithViews();

}

public void Configure(IApplicationBuilder app, IWebHostEnvironment env)

{

    app.UseStaticFiles();

    app.UseRouting();

    app.UseSession(); // <-- ADD THIS (Before UseEndpoints)

    app.UseEndpoints(endpoints =>

    {

        endpoints.MapControllerRoute(

            name: "default",

            pattern: "{controller=Home}/{action=Index}/{id?}");

    });

}
2. Controllers/HomeController.cs
using Microsoft.AspNetCore.Http;

using Microsoft.AspNetCore.Mvc;

namespace YourProjectName.Controllers

{

    public class HomeController : Controller

    {

        public IActionResult Index() => View();

        public IActionResult SetSessionData()

        {

            HttpContext.Session.SetString("Name", "Alex");

            return View();

        }

        public IActionResult GetSessionData()

        {

            ViewBag.Msg = HttpContext.Session.GetString("Name") ?? "No Session Found";

            return View();

        }

        public IActionResult ClearSession()

        {

            HttpContext.Session.Clear();

            return RedirectToAction("GetSessionData");

        }

        public IActionResult SetTempData()

        {

            TempData["Msg"] = "Temporary Message (Disappears on refresh!)";

            return View();

        }

        public IActionResult GetTempData() => View();

    }

}

using System;

class Program
{
    static void Main()
    {
        int n, i;
        bool prime = true;

        Console.Write("Enter a number: ");
        n = Convert.ToInt32(Console.ReadLine());

        if (n <= 1)
            prime = false;
        else
        {
            for (i = 2; i <= Math.Sqrt(n); i++)
            {
                if (n % i == 0)
                {
                    prime = false;
                    break;
                }
            }
        }

        if (prime)
            Console.WriteLine("Prime Number");
        else
            Console.WriteLine("Not a Prime Number");

        Console.ReadLine();
    }
}
